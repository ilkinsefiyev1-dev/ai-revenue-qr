function updateAIRecommendation() { const aiBox = document.getElementById("aiRecommendation"); const hasBurger = cart.some(item => item.name.toLowerCase().includes("chicken burger") ); const hasFries = cart.some(item => item.name.toLowerCase().includes("fries") ); const hasCola = cart.some(item => item.name.toLowerCase().includes("cola") ); const hasCombo = cart.some(item => item.name.toLowerCase().includes("combo") ); if ( hasBurger && !hasFries && !hasCola && !hasCombo ) { aiBox.style.display = "block"; } else { aiBox.style.display = "none"; } }
